# CMDB 研发看板

> [!warning] 只读视图
> 本页由 ZCode 从 Git/GitHub 自动同步，请勿手工修改。卡片点击可跳转对应工单。

```dataviewjs
// 按状态机分列渲染工单卡片；数据源为 plan/ 下的只读投影
const COLUMNS = [
  ["waiting_approval", "待审批"],
  ["ready", "待启动"],
  ["planning", "规划中"],
  ["doing", "开发中"],
  ["testing", "测试中"],
  ["review", "评审中"],
  ["pr_open", "待创建PR"],
  ["pr_checking", "PR检查中"],
  ["waiting_human_merge", "待人工合并"],
  ["waiting_tag_confirm", "待确认Tag"],
  ["building", "构建中"],
  ["waiting_close", "待关闭Issue"],
  ["blocked", "已阻塞"],
  ["done", "已完成"],
];
const DONE_LIMIT = 10;

const pages = [
  ...dv.pages('"plan/01_Requirements"'),
  ...dv.pages('"plan/02_Bugs"'),
];

const byStatus = {};
for (const p of pages) {
  const s = String(p.status ?? "unknown");
  (byStatus[s] ??= []).push(p);
}

const openNote = (path) => () => app.workspace.openLinkText(path, "", false);

const board = dv.container.createDiv();
board.style.display = "flex";
board.style.gap = "10px";
board.style.alignItems = "flex-start";
board.style.overflowX = "auto";

for (const [status, label] of COLUMNS) {
  let items = byStatus[status] ?? [];
  if (status === "done") {
    items = [...items].sort((a, b) => (b.file.mtime?.ts ?? 0) - (a.file.mtime?.ts ?? 0));
  }
  const shown = status === "done" ? items.slice(0, DONE_LIMIT) : items;

  const col = board.createDiv();
  col.style.minWidth = "210px";
  col.style.flex = "1";
  col.style.background = "var(--background-secondary)";
  col.style.borderRadius = "8px";
  col.style.padding = "8px";

  const head = col.createDiv();
  head.style.fontWeight = "600";
  head.style.marginBottom = "6px";
  head.setText(`${label} · ${items.length}`);

  if (shown.length === 0) {
    const empty = col.createDiv();
    empty.style.opacity = "0.5";
    empty.setText("—");
  }
  for (const p of shown) {
    const card = col.createDiv();
    card.style.background = "var(--background-primary)";
    card.style.border = "1px solid var(--background-modifier-border)";
    card.style.borderRadius = "6px";
    card.style.padding = "6px 8px";
    card.style.marginBottom = "6px";
    card.style.fontSize = "0.85em";

    const link = card.createEl("a");
    link.text = `${p.id ?? p.file.name} ${p.title ?? ""}`.trim();
    link.href = p.file.path;
    link.addClass("internal-link");
    link.onclick = openNote(p.file.path);
    link.style.textDecoration = "none";
    link.style.fontWeight = "600";

    const meta = [];
    if (p.github_issue) meta.push(`#${p.github_issue}`);
    if (p.agent_owner) meta.push(String(p.agent_owner));
    if (p.next_action) meta.push(String(p.next_action));
    if (meta.length) {
      const m = card.createDiv();
      m.style.opacity = "0.7";
      m.setText(meta.join(" · "));
    }
  }
  if (status === "done" && items.length > shown.length) {
    const more = col.createDiv();
    more.style.opacity = "0.6";
    more.setText(`… 还有 ${items.length - shown.length} 项`);
  }
}
```
